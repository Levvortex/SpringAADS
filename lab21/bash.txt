#!/bin/bash

print_help() {
    cat << EOF
Usage: solution.sh [OPTION]...
Delete all files with a specified suffix that have not been accessed for a specified number of days.

Mandatory arguments to long options are mandatory for short options too.
  -s, --suffix SUFFIX        suffix of the files to be deleted
  -d, --days DAYS            number of days since last access
  -p, --path PATH            directory to search in (default: .)
  -h, --help                 display this help and exit
EOF
}

SUFFIX=""
DAYS=""
PATH_DIR="."

while [[ "$#" -gt 0 ]]; do
    case $1 in
        -h|--help) 
            print_help
            exit 0 
            ;;
        -s|--suffix) 
            SUFFIX="$2"
            shift 2 
            ;;
        -d|--days) 
            DAYS="$2"
            shift 2 
            ;;
        -p|--path) 
            PATH_DIR="$2"
            shift 2 
            ;;
        *) 
            print_help
            exit 0 
            ;;
    esac
done

if [[ -z "$SUFFIX" || -z "$DAYS" ]]; then
    print_help
    exit 0
fi

find "$PATH_DIR" -type f -name "*${SUFFIX}" -atime +${DAYS} -exec rm -f {} + 2>/dev/null
